        var lg = c.createGain(); lg.gain.value = 0.13; lfo.connect(lg); lg.connect(gain.gain);
        var lfo2 = c.createOscillator(); lfo2.type = 'sine'; lfo2.frequency.value = 0.05;
        var lf = c.createGain(); lf.gain.value = 260; lfo2.connect(lf); lf.connect(filter.frequency);
        lfo.start(); lfo2.start();
        lfos = [lfo, lfo2];
      }
      filter.connect(gain); gain.connect(c.destination);
      src.start();
      amb = { src: src, gain: gain, lfos: lfos };
    }

      if (which === current) return;
      current = which;
      if (which === 'off') stopAmbient(); else startAmbient(which);
      updateUI();
    }
