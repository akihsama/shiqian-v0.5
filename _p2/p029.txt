      particle.style.height = particle.style.width;
      particle.style.background = Math.random() > 0.5 ? 'var(--gold)' : 'var(--text-secondary)';
      
      setTimeout(() => {
        particle.classList.add('active');
      }, Math.random() * 500);
      
      container.appendChild(particle);
    }
  }

  function openStick(event) {
    event.stopPropagation();
    if (isStickOpened) return;
    isStickOpened = true;

    const stickSealed = document.getElementById('stickSealed');
    const stickCard = document.getElementById('stickCard');
    const flyingStick = document.getElementById('flyingStick');

    stickSealed.classList.add('hidden');
    
    setTimeout(() => {
      stickCard.classList.add('unfold');
      flyingStick.classList.add('opened');
      addHistory(window.currentFood);
      // v0.5：翻开签的同时，去平台找这道菜的真实商品
      loadCardPlatform(window.currentFood);
    }, 300);
  }
