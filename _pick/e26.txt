
    if (candidate) {

      if (exact) return exact;
      // 平台真实推荐不在本地签库里，合成一支临时签，保证卡片能显示
      return {
        name: candidate,
        emoji: payload?.recommend?.emoji || '🍽️',
        desc: payload?.recommend?.reason || '食神从附近真实商品里挑的，点下面的按钮直接去平台看。',
        fortune: payload?.recommend?.fortune || '吉',
        blessing: payload?.recommend?.blessing || '顺心顺口，刚刚好',
        ephemeral: true
      };
    }

    return foodDatabase.find(food => reply && reply.includes(food.name));
  }

  async function requestFoodgod() {
    const ctx = window.ShiqianPrefs ? window.ShiqianPrefs.current() : null;
    const pane = window.ShiqianPane && window.ShiqianPane.state;

    const currentFood = window.currentFood
      ? { name: window.currentFood.name, desc: window.currentFood.desc }
      : null;

    return window.ShiqianApi.chat({
      messages: conversation.slice(-AGENT_CONFIG.maxHistoryMessages),
      timeoutMs: AGENT_CONFIG.timeoutMs,
      context: {
        city: ctx ? ctx.city.name : '',
        lat: ctx ? ctx.lat : undefined,
        lng: ctx ? ctx.lng : undefined,
        currentFood,
        // 附近已有平台数据时只传前 20 条，避免 prompt 过长（省钱）
        nearby: pane ? (pane.products || []).slice(0, 20).map(p => ({
          name: p.title,
          price: p.price,
          shop: p.shopName,
          source: p.source
        })) : [],
        foods: foodDatabase.map(food => ({ name: food.name, desc: food.desc }))
      }
    });
  }
