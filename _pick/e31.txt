      } catch(e){}
    }
    function noiseBuffer(type){
      var c = ensureCtx(), len = c.sampleRate * 2;
      var buf = c.createBuffer(1, len, c.sampleRate);
      var d = buf.getChannelData(0);
      if (type === 'rain'){
        for (var i=0;i<len;i++){ d[i] = Math.random()*2-1; }
      } else {
        var last = 0;
        for (var j=0;j<len;j++){ var w = Math.random()*2-1; last = (last + 0.02*w)/1.02; d[j] = last*3.5; }
      }
      return buf;
    }
    function stopAmbient(){
      if (!amb) return;
      var c = ensureCtx(), a = amb; amb = null;
      try {
        var t = c.currentTime;
        a.gain.gain.cancelScheduledValues(t);
        a.gain.gain.setValueAtTime(a.gain.gain.value, t);
        a.gain.gain.linearRampToValueAtTime(0, t + 0.6);
        setTimeout(function(){
          try{ a.src.stop(); }catch(e){}
          if (a.lfos){ try{a.lfos[0].stop();}catch(e){} try{a.lfos[1].stop();}catch(e){} }
        }, 700);
      } catch(e){}
    }
    function startAmbient(which){
      stopAmbient();
      var c = ensureCtx();
      var src = c.createBufferSource(); src.buffer = noiseBuffer(which); src.loop = true;
      var filter = c.createBiquadFilter(); filter.type = 'lowpass';
      var gain = c.createGain(); gain.gain.setValueAtTime(0, c.currentTime);
      var lfos = null;
      if (which === 'rain'){
        var hp = c.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 400;
        filter.frequency.value = 2000;
        src.connect(hp); hp.connect(filter);
        gain.gain.linearRampToValueAtTime(0.42, c.currentTime + 1.2);
      } else {
        filter.frequency.value = 500;
        src.connect(filter);
        gain.gain.linearRampToValueAtTime(0.34, c.currentTime + 1.2);
        var lfo = c.createOscillator(); lfo.type = 'sine'; lfo.frequency.value = 0.08;
        var lg = c.createGain(); lg.gain.value = 0.13; lfo.connect(lg); lg.connect(gain.gain);
