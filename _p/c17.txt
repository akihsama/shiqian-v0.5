      document.getElementById('agentBadge').textContent = payload.model || AGENT_CONFIG.model;
      document.getElementById('agentHint').textContent = payload.searchUsed
        ? '本条回复已联网核对附近真实门店与团购。'
        : '本条回复基于城市、口味与附近平台数据生成（未联网搜索）。';
      conversation.push({ role: 'assistant', content: reply });

      if (food) {
        addHistory(food);
        // 把 AI 选的那道菜作为关键词，刷新下方平台面板
        if (window.ShiqianPane) window.ShiqianPane.searchByFood(food.name);
      }
    } catch (error) {
      console.warn('食神后端调用失败：', error);

      if (!AGENT_CONFIG.fallbackToLocal) {
        renderAgentReply('暂时无法连接食神，请稍后重试。', '连接失败', 'local');
      } else {
        const reply = generateRecommendation(text);
        const food = resolveFood(null, reply);

        renderAgentReply(reply, '本地食神', 'local', '降级模式');
        document.getElementById('agentBadge').textContent = '本地降级';
        document.getElementById('agentHint').textContent = (error && error.code === 'UPSTREAM_BUSY')
          ? '食神通道排队中（免费额度并发限制），这次用本地推荐顶上，过一会儿会自动恢复。'
          : '后端暂不可用，当前使用本地推荐。配置好后会自动切换。';
        conversation.push({ role: 'assistant', content: reply });

        if (food) addHistory(food);
      }
    } finally {
      isSending = false;
      setAgentLoading(false);
    }
  }

  function generateRecommendation(input) {
    const lower = input.toLowerCase();
    const wantsSpicy = /(辣|麻辣|刺激)/.test(lower) && !/(不辣|忌辣|免辣|怕辣|微辣)/.test(lower);
    const preferredNames = new Set();

    const preferenceRules = [
      { pattern: /(清淡|健康|减脂|低脂|素|养胃)/, foods: ['海鲜粥', '越南河粉', '寿司拼盘', '抹茶甜品', '小笼包'] },
      { pattern: /(日式|日本|寿司|咖喱|鳗鱼)/, foods: ['寿司拼盘', '鳗鱼饭', '章鱼小丸子', '日式咖喱饭'] },
      { pattern: /(肉|大份|饱腹|烤肉|牛肉|烤鸭|炸鸡)/, foods: ['烤肉套餐', '红烧牛肉面', '北京烤鸭', '炸鸡配啤酒'] },
      { pattern: /(甜|点心|下午茶|甜品|蛋糕)/, foods: ['抹茶甜品', '提拉米苏', '章鱼小丸子'] },
      { pattern: /(快|简单|方便|上班|赶时间)/, foods: ['蛋包饭', '日式咖喱饭', '小笼包'] },
      { pattern: /(暖|汤|热乎|天气冷|暖胃)/, foods: ['红烧牛肉面', '海鲜粥', '越南河粉', '日式咖喱饭'] },
      { pattern: /(面|面条|粉)/, foods: ['红烧牛肉面', '意大利面', '酸辣粉', '越南河粉'] }
    ];

    preferenceRules.forEach(rule => {
      if (rule.pattern.test(lower)) {
        rule.foods.forEach(name => preferredNames.add(name));
      }
    });

    let candidates = foodDatabase.filter(food => {
      if (/(不辣|忌辣|免辣|怕辣)/.test(lower) && /(辣|麻)/.test(food.name)) return false;
      if (wantsSpicy) return /(辣|麻)/.test(food.name);
      return preferredNames.size === 0 || preferredNames.has(food.name);
    });
