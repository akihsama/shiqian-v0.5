        desc: payload?.recommend?.reason || '食神从附近真实商品里挑的，点下面的按钮直接去平台看。',
        fortune: payload?.recommend?.fortune || '吉',
        blessing: payload?.recommend?.blessing || '顺心顺口，刚刚好',
        ephemeral: true
      };
    }

    return foodDatabase.find(food => reply && reply.includes(food.name));
  }

  async function requestFoodgod() {
    const ctx = window.ShiqianPrefs ? window.ShiqianPrefs.current() : null;
    const pane = window.ShiqianPane && window.ShiqianPane.state;

    const currentFood = window.currentFood
      ? { name: window.currentFood.name, desc: window.currentFood.desc }
      : null;

    return window.ShiqianApi.chat({
      messages: conversation.slice(-AGENT_CONFIG.maxHistoryMessages),
      timeoutMs: AGENT_CONFIG.timeoutMs,
      context: {
        city: ctx ? ctx.city.name : '',
        lat: ctx ? ctx.lat : undefined,
        lng: ctx ? ctx.lng : undefined,
        currentFood,
        // 附近已有平台数据时只传前 20 条，避免 prompt 过长（省钱）
        nearby: pane ? (pane.products || []).slice(0, 20).map(p => ({
          name: p.title,
          price: p.price,
          shop: p.shopName,
          source: p.source
        })) : [],
        foods: foodDatabase.map(food => ({ name: food.name, desc: food.desc }))
      }
    });
  }

  function renderAgentReply(reply, sourceLabel, sourceType, detail = '', sources = []) {
    const result = document.getElementById('llmResult');
    const source = document.createElement('div');
    const content = document.createElement('div');

    source.className = 'agent-reply-source';
    source.textContent = detail ? `${sourceLabel} · ${detail}` : sourceLabel;
    content.textContent = normalizeReply(reply);

    result.replaceChildren(source, content);

    // 联网搜索结果作为小字来源列表，让用户知道这句话有出处
    if (Array.isArray(sources) && sources.length) {
      const box = document.createElement('div');
      box.className = 'agent-sources';
      box.appendChild(document.createTextNode('参考：'));
      sources.slice(0, 3).forEach((s, i) => {
        const a = document.createElement('a');
        a.href = s.url || '#';
        a.target = '_blank';
        a.rel = 'noopener nofollow';
        a.textContent = `${i + 1}. ${s.title || s.site || '来源'}`;
        box.appendChild(a);
      });
      result.appendChild(box);
    }

    result.dataset.source = sourceType;
    result.classList.add('show');
  }

    const text = input.value.trim();

    if (!text || isSending) return;

    input.value = '';
    setAgentLoading(true);
    conversation.push({ role: 'user', content: text });

        payload.model || AGENT_CONFIG.model,
        payload.sources || []
