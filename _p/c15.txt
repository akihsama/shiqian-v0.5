    }
  }

  function closeOverlay() {
    const overlay = document.getElementById('immersiveOverlay');
    const container = document.getElementById('immersiveContainer');
    
    overlay.classList.add('closing');
    container.classList.remove('show');
    
    setTimeout(() => {
      overlay.classList.remove('active', 'closing');
      container.classList.remove('show', 'shaking');
      document.getElementById('flyingStick').classList.remove('fly-out', 'opened');
      document.getElementById('stickSealed').classList.remove('hidden');
      document.getElementById('stickCard').classList.remove('unfold');
      const cp = document.getElementById('cardPlatform');
      if (cp) { cp.hidden = true; cp.innerHTML = ''; }
      document.getElementById('glowEffect').classList.remove('active');
      document.getElementById('closeHint').classList.remove('show');
      isStickOpened = false;
    }, 500);
  }

    }
  }

  function setAgentLoading(isLoading) {
    const input = document.getElementById('llmInput');
    const sendButton = document.getElementById('llmSendBtn');
    const thinking = document.getElementById('llmThinking');
    const result = document.getElementById('llmResult');

    input.disabled = isLoading;
    sendButton.disabled = isLoading;
    sendButton.textContent = isLoading ? '思考中' : '发送';
    thinking.classList.toggle('show', isLoading);

    if (isLoading) {
      result.classList.remove('show');
    } else {
      input.focus({ preventScroll: true });
    }
  }

  function extractAgentReply(payload) {
    if (typeof payload === 'string') return payload.trim();

    const candidates = [
      payload?.reply,
      payload?.content,
      typeof payload?.message === 'string' ? payload.message : payload?.message?.content,
      payload?.choices?.[0]?.message?.content
    ];

    const reply = candidates.find(value => typeof value === 'string' && value.trim());
    return reply ? reply.trim() : '';
  }

  function normalizeReply(reply) {
    return String(reply)
      .replace(/\*\*(.*?)\*\*/g, '$1')
      .replace(/^#{1,6}\s*/gm, '')
      .trim();
  }
